#### 2. 添加 CSRF 保护

**问题**：
- 当前没有 CSRF Token 验证
- 容易受到跨站请求伪造攻击

**解决方案**：

```python
# backend/api/middleware/csrf_middleware.py
from fastapi import Request, HTTPException
from starlette.middleware.base import BaseHTTPMiddleware
import secrets

class CSRFMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        if request.method in ["POST", "PUT", "DELETE", "PATCH"]:
            csrf_token = request.headers.get("X-CSRF-Token")
            session_token = request.session.get("csrf_token")
            
            if not csrf_token or csrf_token != session_token:
                raise HTTPException(status_code=403, detail="CSRF token missing or invalid")
        
        response = await call_next(request)
        return response
```

**优先级**: P0  
**工作量**: 4-6 小时  
**影响**: 防止 CSRF 攻击

**当前问题**：

```python
# backend/api/core/password_policy.py
import re
from datetime import datetime, timedelta

class PasswordPolicy:
    MIN_LENGTH = 12
    REQUIRE_UPPERCASE = True
    REQUIRE_LOWERCASE = True
    REQUIRE_DIGIT = True
    REQUIRE_SPECIAL = True
    PASSWORD_EXPIRY_DAYS = 90
    PASSWORD_HISTORY_COUNT = 5
    
    @staticmethod
    def validate_password(password: str) -> tuple[bool, str]:
        """パスワードポリシーを検証"""
        if len(password) < PasswordPolicy.MIN_LENGTH:
            return False, f"パスワードは{PasswordPolicy.MIN_LENGTH}文字以上である必要があります"
        
        if PasswordPolicy.REQUIRE_UPPERCASE and not re.search(r'[A-Z]', password):
            return False, "パスワードには大文字を含める必要があります"
        
        if PasswordPolicy.REQUIRE_LOWERCASE and not re.search(r'[a-z]', password):
            return False, "パスワードには小文字を含める必要があります"
        
        if PasswordPolicy.REQUIRE_DIGIT and not re.search(r'\d', password):
            return False, "パスワードには数字を含める必要があります"
        
        if PasswordPolicy.REQUIRE_SPECIAL and not re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
            return False, "パスワードには特殊文字を含める必要があります"
        
        return True, "OK"
```

**优先级**: P1  
**工作量**: 6-8 小时  
**影响**: 提高账户安全性

```python
# backend/api/core/cache.py
from functools import lru_cache
from cachetools import TTLCache
import asyncio

class AsyncTTLCache:
    def __init__(self, maxsize=100, ttl=300):
        self.cache = TTLCache(maxsize=maxsize, ttl=ttl)
        self.lock = asyncio.Lock()
    
    async def get(self, key):
        async with self.lock:
            return self.cache.get(key)
    
    async def set(self, key, value):
        async with self.lock:
            self.cache[key] = value

# 使用例
session_cache = AsyncTTLCache(maxsize=1000, ttl=600)  # 10分間キャッシュ

async def get_session_list_cached(user_id: str):
    cache_key = f"sessions:{user_id}"
    cached = await session_cache.get(cache_key)
    if cached:
        return cached
    
    result = await get_session_list(user_id)
    await session_cache.set(cache_key, result)
    return result
```

```python
# backend/api/services/export_service.py
class ExportService:
    @staticmethod
    async def export_session(session_id: str, format: str = "markdown"):
        """セッションをエクスポート"""
        session_data = await get_session_history(session_id)
        
        if format == "markdown":
            return ExportService._to_markdown(session_data)
        elif format == "pdf":
            return await ExportService._to_pdf(session_data)
        elif format == "html":
            return ExportService._to_html(session_data)
        elif format == "docx":
            return await ExportService._to_docx(session_data)
        else:
            return session_data  # JSON
    
    @staticmethod
    def _to_markdown(session_data: dict) -> str:
        """Markdown形式に変換"""
        md = f"# {session_data['session']['title']}\n\n"
        md += f"**作成日時**: {session_data['session']['created_at']}\n\n"
        md += "---\n\n"
        
        for msg in session_data['messages']:
            role = "👤 ユーザー" if msg['role'] == 'user' else "🤖 アシスタント"
            md += f"### {role}\n\n"
            md += f"{msg['content']}\n\n"
            
            if msg.get('source_documents'):
                md += "**参照ドキュメント**:\n"
                for doc in msg['source_documents']:
                    md += f"- {doc.get('metadata', {}).get('filename', 'Unknown')}\n"
                md += "\n"
        
        return md
```

```typescript
// frontend/src/i18n/config.ts
import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';

i18n
  .use(initReactI18next)
  .init({
    resources: {

      },

        translation: require('./locales/ja.json')

        translation: require('./locales/zh.json')

    fallbackLng: 'en',
```
